void FiveQubitCode::applyCorrection(const uint16_t i, double *cArr7)
{                     // an corresponds to Sn stabiliser
    switch (i)        //  X  X  X  Z  Z  Z - stabilizer types. First single Z errors, then X, then Y
    {                 // a1 a2 a3 a4 a5 a6 : Pi where P   {X,Y,Z} is a single-qubit Pauli operator and i is the index of the qubit to correct with i   {0,1,2,3,4};
    case UINT16_C(1): //  0  0  0  0  0  1 : Z4
        FastPauli::Z_r(cArr7, NUMQUBITS, 4);
        break;
    case UINT16_C(2): //  0  0  0  0  1  0 : Z2
        FastPauli::Z_r(cArr7, NUMQUBITS, 2);
        break;
    case UINT16_C(3): //  0  0  0  0  1  1 : Z6
        FastPauli::Z_r(cArr7, NUMQUBITS, 6);
        break;
    case UINT16_C(4): //  0  0  0  1  0  0 : Z1
        FastPauli::Z_r(cArr7, NUMQUBITS, 1);
        break;
    case UINT16_C(5): //  0  0  0  1  0  1 : Z5
        FastPauli::Z_r(cArr7, NUMQUBITS, 5);
        break;
    case UINT16_C(6): //  0  0  0  1  1  0 : Z3
        FastPauli::Z_r(cArr7, NUMQUBITS, 3);
        break;
    case UINT16_C(7): //  0  0  0  1  1  1 : Z7
        FastPauli::Z_r(cArr7, NUMQUBITS, 7);
        break;
    case UINT16_C(8): //  0  0  1  0  0  0 : X4
        FastPauli::X_r(cArr7, NUMQUBITS, 4);
        break;
    case UINT16_C(16): //  0  1  0  0  0  0 : X2
        FastPauli::X_r(cArr7, NUMQUBITS, 2);
        break;
    case UINT16_C(24): //  0  1  1  0  0  0 : X6
        FastPauli::X_r(cArr7, NUMQUBITS, 6);
        break;
    case UINT16_C(32): //  1  0  0  0  0  0 : X1
        FastPauli::X_r(cArr7, NUMQUBITS, 1);
        break;
    case UINT16_C(40): //  1  0  1  0  0  0  : X5
        FastPauli::X_r(cArr7, NUMQUBITS, 5);
        break;
    case UINT16_C(48): //  1  1  0  0  0  0 : X3
        FastPauli::X_r(cArr7, NUMQUBITS, 3);
        break;
    case UINT16_C(56): //  1  1  1  0  0  0 : X7
        FastPauli::X_r(cArr7, NUMQUBITS, 7);
        break;
    case UINT16_C(9): //  0  0  1  0  0  1 : Y4
        FastPauli::Y_r(cArr7, NUMQUBITS, 4);
        break;
    case UINT16_C(18): //  0  1  0  0  1  0 : Y2
        FastPauli::Y_r(cArr7, NUMQUBITS, 2);
        break;
    case UINT16_C(27): //  0  1  1  0  1  1 : Y6
        FastPauli::Y_r(cArr7, NUMQUBITS, 6);
        break;
    case UINT16_C(36): //  1  0  0  1  0  0 : Y1
        FastPauli::Y_r(cArr7, NUMQUBITS, 1);
        break;
    case UINT16_C(45): //  1  0  1  1  0  1 : Y5
        FastPauli::Y_r(cArr7, NUMQUBITS, 5);
        break;
    case UINT16_C(54): //  1  1  0  1  1  0 : Y3
        FastPauli::Y_r(cArr7, NUMQUBITS, 3);
        break;
    case UINT16_C(63): //  1  1  1  1  1  1 : Y7
        FastPauli::Y_r(cArr7, NUMQUBITS, 7);
        break;

    default:
        break;
    }
}
